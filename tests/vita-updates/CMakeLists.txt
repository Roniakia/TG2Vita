cmake_minimum_required(VERSION 3.16)
set(CMAKE_TOOLCHAIN_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../../cmake/vita-toolchain.cmake")
project(vita_update_smoke VERSION 0.0.0 LANGUAGES CXX C)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
include("${VITASDK}/share/vita.cmake")
get_filename_component(APP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(VITA_MAKE_FSELF "${APP_ROOT}/scripts/vita-host-tool.sh;vita-make-fself")
set(VITA_PACK_VPK "${APP_ROOT}/scripts/vita-host-tool.sh;vita-pack-vpk")
configure_file("${APP_ROOT}/src/version.hpp.in" generated/version.hpp @ONLY)
set(JANSSON_WITHOUT_TESTS ON CACHE BOOL "" FORCE)
set(JANSSON_BUILD_DOCS OFF CACHE BOOL "" FORCE)
set(JANSSON_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${APP_ROOT}/.deps/jansson-2.15.1" jansson)
set_target_properties(jansson PROPERTIES POSITION_INDEPENDENT_CODE OFF)
add_executable(vita_update_smoke main.cpp "${APP_ROOT}/src/update/updater.cpp")
target_compile_features(vita_update_smoke PRIVATE cxx_std_17)
target_include_directories(vita_update_smoke PRIVATE "${APP_ROOT}/src" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_options(vita_update_smoke PRIVATE -Wl,--gc-sections -Wl,--pic-veneer -Wl,-u,pthread_cancel -Wl,-u,pthread_once)
target_link_libraries(vita_update_smoke PRIVATE jansson curl ssl crypto zstd z pthread SceNet_stub SceNetCtl_stub SceSysmodule_stub SceIofilemgr_stub SceKernelThreadMgr_stub m)
vita_create_self(vita_update_smoke.self vita_update_smoke)
vita_create_vpk(vita_update_smoke.vpk VTGC00004 vita_update_smoke.self VERSION "00.10" NAME "Vita TG Update Tests" FILE "${APP_ROOT}/assets/certs/cacert.pem" assets/certs/cacert.pem)
